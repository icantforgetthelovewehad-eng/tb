local function addSnowfall(parent, amount, zIndex)
    local layer = Instance.new("Frame")

    layer.Name = "Snowfall"
    layer.Size = UDim2.fromScale(1, 1)
    layer.BackgroundTransparency = 1
    layer.BorderSizePixel = 0
    layer.ClipsDescendants = true
    layer.ZIndex = zIndex or 1
    layer.Active = false
    layer.Selectable = false
    layer.Parent = parent

    local rng = Random.new()

    for _ = 1, amount do
        local flake = Instance.new("Frame")
        local size = rng:NextInteger(2, 5)

        flake.Name = "Snowflake"
        flake.Size = UDim2.fromOffset(size, size)

        flake.Position = UDim2.new(
            rng:NextNumber(),
            0,
            rng:NextNumber(-0.15, 1),
            0
        )

        local shade = rng:NextInteger(205, 245)

        flake.BackgroundColor3 = Color3.fromRGB(
            shade,
            shade,
            shade
        )

        flake.BackgroundTransparency =
            rng:NextNumber(0.18, 0.58)

        flake.BorderSizePixel = 0
        flake.ZIndex = layer.ZIndex
        flake.Active = false
        flake.Selectable = false

        Instance.new("UICorner", flake).CornerRadius =
            UDim.new(1, 0)

        flake.Parent = layer

        task.spawn(function()

            while flake.Parent do

                local startX =
                    rng:NextNumber(-0.08, 1.08)

                local duration =
                    rng:NextNumber(4.5, 9.5)

                local drift =
                    rng:NextNumber(-0.07, 0.07)

                flake.Position = UDim2.new(
                    startX,
                    0,
                    -0.08,
                    0
                )

                local tween = TweenService:Create(
                    flake,
                    TweenInfo.new(
                        duration,
                        Enum.EasingStyle.Linear
                    ),
                    {
                        Position = UDim2.new(
                            startX + drift,
                            0,
                            1.08,
                            0
                        )
                    }
                )

                tween:Play()
                tween.Completed:Wait()

                if flake.Parent then
                    task.wait(
                        rng:NextNumber(0, 0.8)
                    )
                end

            end

        end)
    end

    return layer
end

local function refreshWhitelistUI()

    for _, child in ipairs(
        wlList:GetChildren()
    ) do

        if child:IsA("TextButton") then
            child:Destroy()
        end

    end

    local query =
        string.lower(
            searchBox.Text or ""
        )

    for _, player in ipairs(
        Players:GetPlayers()
    ) do

        if player ~= LocalPlayer then

            local displayName =
                player.DisplayName

            local username =
                player.Name

            local matches =
                query == ""
                or string.find(
                    string.lower(displayName),
                    query,
                    1,
                    true
                )
                or string.find(
                    string.lower(username),
                    query,
                    1,
                    true
                )

            if matches then

                local enabled =
                    WHITELIST[player] == true

                local row =
                    Instance.new(
                        "TextButton",
                        wlList
                    )

                row.Name =
                    "Player_" .. player.Name

                row.Size =
                    UDim2.new(1, 0, 0, 36)

                row.BackgroundColor3 =
                    enabled
                    and Color3.fromRGB(35, 35, 35)
                    or Color3.fromRGB(15, 15, 15)

                row.BackgroundTransparency = 0.12
                row.BorderSizePixel = 0
                row.AutoButtonColor = false
                row.Text = ""
                row.ZIndex = 12

                Instance.new(
                    "UICorner",
                    row
                ).CornerRadius =
                    UDim.new(0, 8)

                local name =
                    Instance.new(
                        "TextLabel",
                        row
                    )

                name.Size =
                    UDim2.new(1, -54, 1, 0)

                name.Position =
                    UDim2.new(0, 12, 0, 0)

                name.BackgroundTransparency = 1

                name.Text =
                    displayName
                    .. (
                        displayName ~= username
                        and "  @" .. username
                        or ""
                    )

                name.TextColor3 =
                    enabled
                    and Color3.fromRGB(252, 252, 252)
                    or Color3.fromRGB(190, 190, 190)

                name.Font =
                    Enum.Font.GothamMedium

                -- Player names unchanged.
                name.TextSize = 11

                name.TextXAlignment =
                    Enum.TextXAlignment.Left

                name.TextYAlignment =
                    Enum.TextYAlignment.Center

                name.TextTruncate =
                    Enum.TextTruncate.AtEnd

                name.ZIndex = 13
                name.Active = false

                -- =================================================
                -- CHECKBOX
                -- =================================================

                local checkbox =
                    Instance.new(
                        "Frame",
                        row
                    )

                checkbox.Name =
                    "Checkbox"

                checkbox.Size =
                    UDim2.fromOffset(18, 18)

                checkbox.Position =
                    UDim2.new(
                        1,
                        -30,
                        0.5,
                        -9
                    )

                checkbox.BackgroundColor3 =
                    enabled
                    and Color3.fromRGB(235, 235, 235)
                    or Color3.fromRGB(25, 25, 25)

                checkbox.BackgroundTransparency =
                    enabled and 0 or 0.15

                checkbox.BorderSizePixel = 0
                checkbox.ZIndex = 14
                checkbox.Active = false

                Instance.new(
                    "UICorner",
                    checkbox
                ).CornerRadius =
                    UDim.new(0, 5)

                local checkboxStroke =
                    Instance.new(
                        "UIStroke",
                        checkbox
                    )

                checkboxStroke.Color =
                    enabled
                    and Color3.fromRGB(255, 255, 255)
                    or Color3.fromRGB(100, 100, 100)

                checkboxStroke.Transparency =
                    enabled and 0.05 or 0.35

                checkboxStroke.Thickness = 1

                local check =
                    Instance.new(
                        "TextLabel",
                        checkbox
                    )

                check.Size =
                    UDim2.fromScale(1, 1)

                check.BackgroundTransparency = 1
                check.Text = "✓"

                check.TextColor3 =
                    Color3.fromRGB(15, 15, 15)

                check.Font =
                    Enum.Font.GothamBold

                check.TextSize = 13

                check.TextXAlignment =
                    Enum.TextXAlignment.Center

                check.TextYAlignment =
                    Enum.TextYAlignment.Center

                check.Visible = enabled
                check.ZIndex = 15
                check.Active = false

                row.MouseEnter:Connect(
                    function()

                        row.BackgroundColor3 =
                            Color3.fromRGB(40, 40, 40)

                    end
                )

                row.MouseLeave:Connect(
                    function()

                        row.BackgroundColor3 =
                            enabled
                            and Color3.fromRGB(35, 35, 35)
                            or Color3.fromRGB(15, 15, 15)

                    end
                )

                row.MouseButton1Click:Connect(
                    function()

                        if WHITELIST[player] then
                            WHITELIST[player] = nil
                        else
                            WHITELIST[player] = true
                        end

                        refreshWhitelistUI()

                    end
                )

            end

        end

    end

end

-- =========================================================
-- DRAGGING + RESIZING
-- =========================================================

local resizeHandle

local dragging = false
local dragStart
local startPosition

local resizing = false
local resizeStart
local originalSize

local MIN_WIDTH = 300
local MIN_HEIGHT = 260

local function pointInsideCard(position)

    local cardPos =
        wlCard.AbsolutePosition

    local cardSize =
        wlCard.AbsoluteSize

    return
        position.X >= cardPos.X
        and position.X <=
            cardPos.X + cardSize.X
        and position.Y >= cardPos.Y
        and position.Y <=
            cardPos.Y + cardSize.Y

end

local function isOverResizeHandle(position)

    if not resizeHandle then
        return false
    end

    local handlePos =
        resizeHandle.AbsolutePosition

    local handleSize =
        resizeHandle.AbsoluteSize

    return
        position.X >= handlePos.X
        and position.X <=
            handlePos.X + handleSize.X
        and position.Y >= handlePos.Y
        and position.Y <=
            handlePos.Y + handleSize.Y

end

-- =========================================================
-- RESIZE HANDLE
-- =========================================================

resizeHandle =
    Instance.new(
        "Frame",
        wlCard
    )

resizeHandle.Name =
    "ResizeHandle"

resizeHandle.Size =
    UDim2.fromOffset(20, 20)

resizeHandle.Position =
    UDim2.new(
        1,
        -20,
        1,
        -20
    )

resizeHandle.BackgroundTransparency = 1
resizeHandle.BorderSizePixel = 0
resizeHandle.ZIndex = 20
resizeHandle.Active = true

-- =========================================================
-- THREE-DOT RESIZE DESIGN
-- =========================================================

for i = 0, 2 do

    local dot =
        Instance.new(
            "Frame",
            resizeHandle
        )

    dot.Name =
        "ResizeDot"

    dot.Size =
        UDim2.fromOffset(4, 4)

    dot.Position =
        UDim2.new(
            0,
            5 + i * 5,
            0,
            15 - i * 5
        )

    dot.BackgroundColor3 =
        Color3.fromRGB(190, 190, 190)

    dot.BackgroundTransparency = 0.15
    dot.BorderSizePixel = 0
    dot.ZIndex = 21
    dot.Active = false

    Instance.new(
        "UICorner",
        dot
    ).CornerRadius =
        UDim.new(1, 0)

end

-- =========================================================
-- START DRAG / RESIZE
-- =========================================================

UserInputService.InputBegan:Connect(
    function(input)

        if input.UserInputType ~=
            Enum.UserInputType.MouseButton1 then
            return
        end

        if isOverResizeHandle(
            input.Position
        ) then

            resizing = true
            resizeStart = input.Position
            originalSize = wlCard.AbsoluteSize

            return
        end

        if pointInsideCard(
            input.Position
        ) then

            dragging = true
            dragStart = input.Position
            startPosition = wlCard.Position

        end

    end
)

-- =========================================================
-- MOUSE MOVEMENT
-- =========================================================

UserInputService.InputChanged:Connect(
    function(input)

        if input.UserInputType ~=
            Enum.UserInputType.MouseMovement then
            return
        end

        if resizing then

            local delta =
                input.Position - resizeStart

            local newWidth =
                math.max(
                    MIN_WIDTH,
                    originalSize.X + delta.X
                )

            local newHeight =
                math.max(
                    MIN_HEIGHT,
                    originalSize.Y + delta.Y
                )

            wlCard.Size =
                UDim2.fromOffset(
                    newWidth,
                    newHeight
                )

            return
        end

        if dragging then

            local delta =
                input.Position - dragStart

            wlCard.Position =
                UDim2.new(
                    startPosition.X.Scale,
                    startPosition.X.Offset
                        + delta.X,

                    startPosition.Y.Scale,
                    startPosition.Y.Offset
                        + delta.Y
                )

        end

    end
)

-- =========================================================
-- END DRAG / RESIZE
-- =========================================================

UserInputService.InputEnded:Connect(
    function(input)

        if input.UserInputType ==
            Enum.UserInputType.MouseButton1 then

            dragging = false
            resizing = false

        end

    end
)

-- =========================================================
-- F4 TOGGLE
-- =========================================================

UserInputService.InputBegan:Connect(
    function(
        input,
        gameProcessed
    )

        if gameProcessed then
            return
        end

        if input.KeyCode ==
            Enum.KeyCode.F4 then

            whitelistGui.Enabled =
                not whitelistGui.Enabled

            if whitelistGui.Enabled then
                refreshWhitelistUI()
            end

        end

    end
)

-- =========================================================

local function getPlayerFromTarget(target)

    if not target then
        return nil
    end

    local model =
        target:FindFirstAncestorOfClass(
            "Model"
        )

    if not model then
        return nil
    end

    return Players:GetPlayerFromCharacter(
        model
    )

end

local lastShot = 0

RunService.RenderStepped:Connect(
    function()

        if not TRIGGER_HELD then
            return
        end

        if tick() - lastShot <
            FIRE_RATE then
            return
        end

        local target =
            Mouse.Target

        local player =
            getPlayerFromTarget(
                target
            )

        if not player
            or player == LocalPlayer then
            return
        end

        if WHITELIST[player] then
            return
        end

        if WHITELIST[player.UserId] then
            return
        end

        mouse1press()

        task.wait(0.02)

        mouse1release()

        lastShot = tick()

    end
)
